library(tidyverse)
library(dplyr)
library(ggplot2)
library(readr)
library(gridExtra)
library(cowplot)
library(patchwork)


############################# BENCHMARK nginx ################################


# Import netperf file

nginx<- readr::read_csv("~/Dicht/nginx/nginx.csv")

# Format column types

nginx$Concurrency <- as.factor(nginx$Concurrency)   

## PLOTS ##

# TP



################################### visualisation debit ###################################


p<-ggplot(nginx, aes(x=Concurrency,Requests_mean, fill=environment,label=Requests_mean)) + 
  geom_bar(stat="identity", color="black",alpha=1,width=.8,
           position=position_dodge()) +
  labs(x="Simultaneous clients", y="Throughput (RPS)")





p<- p +theme_classic()+ theme(legend.position = "top",legend.title = element_blank(),
                                 legend.text = element_text( size = 44),
                                 axis.text.x  = element_text(vjust=0.5, size=44,angle=60,color = "black"),
                                 axis.text.y  = element_text(vjust=0.5, size=44,color = "black"),
                                 axis.title.x = element_text(vjust=0.5, size=44,color = "black"),
                                 axis.title.y = element_text(vjust=0.5, size=50,color = "black"))

p<-p + scale_x_discrete(labels=c("1000" = "1k", "2000" = "2k",
                                   "3000" = "3k","4000" = "4k", "5000" = "5k",
                                   "6000" = "6k","7000" = "7k", "8000" = "8k",
                                   "9000" = "9k"))

p

ggsave("nginx.pdf",width=26,height = 10)
