library(tidyverse)
library(dplyr)
library(ggplot2)
library(readr)
library(gridExtra)
library(cowplot)
library(patchwork)


############################# BENCHMARK scalability POSTGRESQL  ################################


# Set Working Directory
setwd("~/Dicht/scalabity/")


# Import netperf file
df<- readr::read_csv("scalabity_final.csv")

# Format column types

df$containers <- as.factor(df$containers)    

## PLOTS ##

############################### visualisation latence #########################


p<- ggplot(df, aes(x=containers, y=time_mean, fill=Environnement,label=time_mean)) + 
  geom_bar(stat="identity", color="black",alpha=1,width=.8,
           position=position_dodge()) +
  geom_errorbar(aes(ymin=time_mean-time_std, ymax=time_mean+time_std), width=.2,
                position=position_dodge(.9)) +
  labs(x="Container running", y="Latency (s)")
 


 p +theme_classic()+theme(legend.position = "top",legend.title = element_blank(),
              legend.text = element_text( size = 40),
              axis.text.x  = element_text(vjust=0.5,size=45,color = "black"),
              axis.text.y  = element_text(vjust=0.5, size=45,color = "black"),
              axis.title.x = element_text(vjust=0.5,size=44,color = "black"),
              axis.title.y = element_text(vjust=0.5, size=50,color = "black"))


ggsave("echelle_new1.pdf",width=25,height = 10)
